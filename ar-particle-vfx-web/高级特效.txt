        // Programmatic Glowing Particle Canvas Texture
        function createParticleTexture() {
            const canvas = document.createElement('canvas');
            canvas.width = 64;
            canvas.height = 64;
            const ctx = canvas.getContext('2d');

            const gradient = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);
            gradient.addColorStop(0, 'rgba(255, 255, 255, 1)');
            gradient.addColorStop(0.2, 'rgba(255, 255, 255, 0.8)');
            gradient.addColorStop(0.5, 'rgba(255, 255, 255, 0.3)');
            gradient.addColorStop(1, 'rgba(255, 255, 255, 0)');

            ctx.fillStyle = gradient;
            ctx.fillRect(0, 0, 64, 64);

            const texture = new THREE.CanvasTexture(canvas);
            return texture;
        }

        // ==========================================
        // 3. PARAMETRIC 3D SHAPE GENERATORS
        // ==========================================
        function generateShapePoints(type) {
            const points = new Float32Array(PARTICLE_COUNT * 3);
            const vels = new Float32Array(PARTICLE_COUNT * 3);

            for (let i = 0; i < PARTICLE_COUNT; i++) {
                let x = 0, y = 0, z = 0;
                const idx = i * 3;

                switch (type) {
                    case 'heart': {
                        // 3D Heart Surface Parameterization
                        const t = Math.PI * (2 * Math.random() - 1);
                        const p = Math.PI * (Math.random() - 0.5);
                        
                        x = 16 * Math.pow(Math.sin(t), 3) * Math.cos(p);
                        y = (13 * Math.cos(t) - 5 * Math.cos(2*t) - 2 * Math.cos(3*t) - Math.cos(4*t));
                        z = 10 * Math.sin(t) * Math.sin(p);
                        
                        // Volume jitter
                        const scale = 0.85 + Math.random() * 0.3;
                        x *= scale * 0.85;
                        y *= scale * 0.85;
                        z *= scale * 0.85;
                        break;
                    }

                    case 'flower': {
                        // 3D Rose Flower Surface
                        const phi = Math.random() * Math.PI * 2;
                        const theta = Math.random() * Math.PI;
                        const k = 5; // Petal multiplier
                        const r = 16 * (1 + 0.35 * Math.cos(k * phi)) * Math.sin(theta);

                        x = r * Math.sin(theta) * Math.cos(phi);
                        y = 16 * Math.cos(theta) + 4 * Math.sin(k * phi);
                        z = r * Math.sin(theta) * Math.sin(phi);
                        break;
                    }

                    case 'saturn': {
                        // Saturn Core Sphere + Outer Rings
                        if (Math.random() < 0.35) {
                            // Core Sphere
                            const u = Math.random();
                            const v = Math.random();
                            const theta = u * 2.0 * Math.PI;
                            const phi = Math.acos(2.0 * v - 1.0);
                            const r = 7 * Math.cbrt(Math.random());

                            x = r * Math.sin(phi) * Math.cos(theta);
                            y = r * Math.sin(phi) * Math.sin(theta);
                            z = r * Math.cos(phi);
                        } else {
                            // Disk Ring
                            const r = 12 + Math.random() * 16;
                            const theta = Math.random() * Math.PI * 2;
                            
                            // Tilted ring
                            const rx = r * Math.cos(theta);
                            const rz = r * Math.sin(theta);
                            const ry = (Math.random() - 0.5) * 0.8;

                            // Tilt ring by ~25 deg
                            const angle = 0.45;
                            x = rx;
                            y = ry * Math.cos(angle) - rz * Math.sin(angle);
                            z = ry * Math.sin(angle) + rz * Math.cos(angle);
                        }
                        break;
                    }

                    case 'buddha': {
                        // Mathematical Silhouette of Meditating Zen Figure
                        const p = Math.random();
                        if (p < 0.20) {
                            // Head
                            const u = Math.random(), v = Math.random();
                            const theta = u * 2 * Math.PI, phi = Math.acos(2 * v - 1);
                            const r = 3.2 * Math.cbrt(Math.random());
                            x = r * Math.sin(phi) * Math.cos(theta);
                            y = 11.5 + r * Math.sin(phi) * Math.sin(theta);
                            z = r * Math.cos(phi);
                        } else if (p < 0.26) {
                            // Usnisa / Crown
                            const u = Math.random(), v = Math.random();
                            const theta = u * 2 * Math.PI, phi = Math.acos(2 * v - 1);
                            const r = 1.4 * Math.cbrt(Math.random());
                            x = r * Math.sin(phi) * Math.cos(theta);
                            y = 15.2 + r * Math.sin(phi) * Math.sin(theta);
                            z = r * Math.cos(phi);
                        } else if (p < 0.65) {
                            // Torso & Shoulders (Pyramidal Taper)
                            const h = Math.random(); // 0 to 1
                            const layerY = -2 + h * 11;
                            const radius = (1.0 - h * 0.55) * 8.5;
                            const angle = Math.random() * Math.PI * 2;
                            const r = Math.sqrt(Math.random()) * radius;

                            x = r * Math.cos(angle);
                            y = layerY;
                            z = r * Math.sin(angle) * 0.75;
                        } else {
                            // Cross-legged Lotus Base
                            const angle = Math.random() * Math.PI * 2;
                            const rx = 14 * Math.sqrt(Math.random());
                            const ry = (Math.random() - 0.5) * 3.5;
                            const rz = 10 * Math.sqrt(Math.random());

                            x = rx * Math.cos(angle);
                            y = -6.5 + ry;
                            z = rz * Math.sin(angle);
                        }
                        break;
                    }

                    case 'fireworks': {
                        // Explosion Sphere Vectors
                        const u = Math.random();
                        const v = Math.random();
                        const theta = u * 2.0 * Math.PI;
                        const phi = Math.acos(2.0 * v - 1.0);
                        const speed = 10 + Math.random() * 25;

                        x = speed * Math.sin(phi) * Math.cos(theta);
                        y = speed * Math.sin(phi) * Math.sin(theta);
                        z = speed * Math.cos(phi);
                        break;
                    }
                }

                points[idx] = x;
                points[idx + 1] = y;
                points[idx + 2] = z;

                // Burst velocity vector
                const mag = Math.sqrt(x*x + y*y + z*z) || 1.0;
                vels[idx] = (x / mag) * (1 + Math.random() * 2);
                vels[idx + 1] = (y / mag) * (1 + Math.random() * 2);
                vels[idx + 2] = (z / mag) * (1 + Math.random() * 2);
            }

            return { points, vels };
        }

        // ==========================================
        // 4. PARTICLE SYSTEM CREATION & UPDATE
        // ==========================================
        function createParticleSystem() {
            particleGeometry = new THREE.BufferGeometry();
            
            const shapeData = generateShapePoints(state.preset);
            targetPositions = shapeData.points;
            initialVelocities = shapeData.vels;

            positions = new Float32Array(PARTICLE_COUNT * 3);
            for (let i = 0; i < PARTICLE_COUNT * 3; i++) {
                positions[i] = targetPositions[i];
            }

            particleGeometry.setAttribute('position', new THREE.BufferAttribute(positions, 3));

            const particleMaterial = new THREE.PointsMaterial({
                color: new THREE.Color(state.color),
                size: 0.85,
                map: createParticleTexture(),
                transparent: true,
                opacity: 0.85,
                blending: THREE.AdditiveBlending,
                depthWrite: false
            });

            particleSystem = new THREE.Points(particleGeometry, particleMaterial);
            scene.add(particleSystem);
        }

        function switchPreset(newPreset) {
            state.preset = newPreset;
            const shapeData = generateShapePoints(newPreset);
            targetPositions = shapeData.points;
            initialVelocities = shapeData.vels;
        }

        let hands, cameraUtils;

        function initHandTracking() {
            hands = new Hands({
                locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
            });

            hands.setOptions({
                maxNumHands: 2,
                modelComplexity: 1,
                minDetectionConfidence: 0.6,
                minTrackingConfidence: 0.6
            });

            hands.onResults(onHandResults);

            cameraUtils = new Camera(webcamVideo, {
                onFrame: async () => {
                    if (state.cameraActive) {
                        await hands.send({ image: webcamVideo });
                    }
                },
                width: 320,
                height: 240
            });
        }

        function onHandResults(results) {
            webcamCtx.save();
            webcamCtx.clearRect(0, 0, webcamCanvas.width, webcamCanvas.height);

            if (results.multiHandLandmarks && results.multiHandLandmarks.length > 0) {
                state.handDetected = true;
                cameraStatusDot.classList.add('active');

                const landmarks = results.multiHandLandmarks;

                // Draw hand landmarks on small HUD preview
                for (const lms of landmarks) {
                    drawLandmarks(webcamCtx, lms);
                }

                if (landmarks.length === 2) {
                    // TWO HANDS: Control Scale via Distance between palms (Landmark 9: Middle Finger MCP)
                    const h1 = landmarks[0][9];
                    const h2 = landmarks[1][9];
                    const dist = Math.hypot(h1.x - h2.x, h1.y - h2.y);
                    
                    // Map distance [0.15, 0.75] to Scale [0.4, 2.8]
                    const mappedScale = Math.min(Math.max((dist - 0.15) * 4.0 + 0.4, 0.3), 3.0);
                    state.scale = THREE.MathUtils.lerp(state.scale, mappedScale, 0.1);
                    
                    sliderScale.value = state.scale.toFixed(2);
                    valScale.textContent = `${state.scale.toFixed(1)}x`;

                    // Combined spread of fingers -> Dispersion
                    const spread1 = getHandSpread(landmarks[0]);
                    const spread2 = getHandSpread(landmarks[1]);
                    const avgSpread = (spread1 + spread2) / 2;

                    state.targetDispersion = Math.min(Math.max((avgSpread - 0.25) * 150, 0), 100);

                    cameraStatusText.textContent = "双手接通 (缩放 + 扩散)";
                    gestureTitle.textContent = `双掌间距: ${(dist * 100).toFixed(0)}cm`;
                    gestureSub.textContent = `当前缩放: ${state.scale.toFixed(2)}x | 扩散: ${state.dispersion.toFixed(0)}`;

                } else if (landmarks.length === 1) {
                    // ONE HAND: Open/Close Palm -> Dispersion Force
                    const lms = landmarks[0];
                    const spread = getHandSpread(lms);

                    // Map palm spread [0.2, 0.55] to dispersion [0, 100]
                    state.targetDispersion = Math.min(Math.max((spread - 0.22) * 220, 0), 100);
                    
                    // Rotate scene slightly based on hand movement (Landmark 9)
                    const palmX = (lms[9].x - 0.5) * 2;
                    particleSystem.rotation.y = THREE.MathUtils.lerp(particleSystem.rotation.y, palmX * 1.5, 0.05);

                    cameraStatusText.textContent = "单手跟踪中";
                    gestureTitle.textContent = spread > 0.4 ? "手掌张开 (粒子扩散)" : "拳头捏合 (凝聚聚集)";
                    gestureSub.textContent = `手势张合度: ${(spread * 100).toFixed(0)}%`;
                }

                sliderDispersion.value = state.targetDispersion;
                valDispersion.textContent = state.targetDispersion.toFixed(0);

            } else {
                state.handDetected = false;
                cameraStatusDot.classList.remove('active');
                cameraStatusText.textContent = "摄像头: 正在扫描双手...";
                gestureTitle.textContent = "未检测到手势";
                gestureSub.textContent = "请将双手放入镜头画面中";
            }

            webcamCtx.restore();
        }

        // Helper: Compute average distance from Wrist (0) to Fingertips (4, 8, 12, 16, 20)
        function getHandSpread(landmarks) {
            const wrist = landmarks[0];
            const tips = [4, 8, 12, 16, 20];
            let totalDist = 0;
            for (const tip of tips) {
                totalDist += Math.hypot(landmarks[tip].x - wrist.x, landmarks[tip].y - wrist.y);
            }
            return totalDist / 5;
        }

        // Lightweight landmark renderer for preview HUD
        function drawLandmarks(ctx, landmarks) {
            ctx.fillStyle = '#ff2a70';
            ctx.strokeStyle = 'rgba(255, 255, 255, 0.6)';
            ctx.lineWidth = 1;

            const w = webcamCanvas.width;
            const h = webcamCanvas.height;

            for (let i = 0; i < landmarks.length; i++) {
                const pt = landmarks[i];
                ctx.beginPath();
                ctx.arc(pt.x * w, pt.y * h, 2, 0, 2 * Math.PI);
                ctx.fill();
            }
        }

        function onWindowResize() {
            camera.aspect = window.innerWidth / window.innerHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(window.innerWidth, window.innerHeight);
        }

        // ==========================================
        // 7. ANIMATION RENDER LOOP
        // ==========================================
        const clock = new THREE.Clock();

        function animate() {
            requestAnimationFrame(animate);

            const elapsedTime = clock.getElapsedTime();

            // Smooth Interpolation of Dispersion Force
            state.dispersion = THREE.MathUtils.lerp(state.dispersion, state.targetDispersion, 0.08);

            // Decay burst force over time
            if (state.burstForce > 0.1) {
                state.burstForce *= 0.92;
            } else {
                state.burstForce = 0;
            }

            const currentDispersion = state.dispersion + state.burstForce;

            // Global Scene Gentle Idle Rotation
            if (!state.handDetected) {
                particleSystem.rotation.y = elapsedTime * 0.12;
            }
            particleSystem.rotation.x = Math.sin(elapsedTime * 0.08) * 0.1;

            // Scale System
            particleSystem.scale.setScalar(THREE.MathUtils.lerp(particleSystem.scale.x, state.scale, 0.08));

            // Particle Position Updates (Morphing + Dispersion Physics)
            const posAttr = particleGeometry.attributes.position;
            const posArray = posAttr.array;

            const lerpSpeed = 0.06; // Smooth shape transition rate

            for (let i = 0; i < PARTICLE_COUNT; i++) {
                const idx = i * 3;

                // Base morphing towards target shape position
                let tx = targetPositions[idx];
                let ty = targetPositions[idx + 1];
                let tz = targetPositions[idx + 2];

                // Calculate dispersion vector outward from center
                if (currentDispersion > 0.1) {
                    const vx = initialVelocities[idx];
                    const vy = initialVelocities[idx + 1];
                    const vz = initialVelocities[idx + 2];

                    const factor = currentDispersion * 0.35;
                    tx += vx * factor;
                    ty += vy * factor;
                    tz += vz * factor;
                }

                // Smooth position lerp
                posArray[idx] += (tx - posArray[idx]) * lerpSpeed;
                posArray[idx + 1] += (ty - posArray[idx + 1]) * lerpSpeed;
                posArray[idx + 2] += (tz - posArray[idx + 2]) * lerpSpeed;
            }

            posAttr.needsUpdate = true;

            controls.update();
            renderer.render(scene, camera);
        }

        // Initialize Everything
        window.addEventListener('DOMContentLoaded', () => {
            initThree();
            animate();
        });
